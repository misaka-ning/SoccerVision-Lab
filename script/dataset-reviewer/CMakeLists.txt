cmake_minimum_required(VERSION 3.21)
project(FootballDatasetReviewer VERSION 1.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
add_library(review-store STATIC src/dataset_store.cpp src/dataset_store.h)
target_link_libraries(review-store PUBLIC Qt6::Core)
target_include_directories(review-store PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_executable(football-dataset-reviewer src/main.cpp src/label_editor.h)
target_link_libraries(football-dataset-reviewer PRIVATE review-store Qt6::Gui Qt6::Widgets)
set_target_properties(football-dataset-reviewer PROPERTIES BUILD_RPATH "${Qt6_DIR}/../../")
add_executable(qt_review_lock_runner src/qt_review_lock_runner.cpp)
target_link_libraries(qt_review_lock_runner PRIVATE Qt6::Core)
set_target_properties(qt_review_lock_runner PROPERTIES BUILD_RPATH "${Qt6_DIR}/../../")
option(BUILD_TESTING "Build CSV and label safety tests" OFF)
if(BUILD_TESTING)
  enable_testing()
  find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
  target_compile_definitions(football-dataset-reviewer PRIVATE REVIEW_GUI_TESTING)
  target_link_libraries(football-dataset-reviewer PRIVATE Qt6::Test)
  add_executable(review-store-tests tests/dataset_store_tests.cpp)
  target_link_libraries(review-store-tests PRIVATE review-store Qt6::Test)
  add_test(NAME review-store-tests COMMAND review-store-tests)
endif()
